function filterPosts(category, clickedButton) {
    const buttons = document.querySelectorAll(".filter");
    const posts = document.querySelectorAll(".post");

    buttons.forEach(function(button) {
        button.classList.remove("active");
    });

    clickedButton.classList.add("active");

    posts.forEach(function(post) {
        if (category === "all" || post.classList.contains(category)) {
            post.style.display = "flex";
        } else {
            post.style.display = "none";
        }
    });
}

function showTip(tip) {
    if (tip === "composition-landscape") {
        alert("Tip: Use the rule of thirds to create a more interesting composition.");
    }
    if (tip === "lighting-landscape") {
        alert("Tip: Use natural light to create a more dramatic effect.");
    }
    if (tip === "perspective-landscape") {
        alert("Tip: Use a low angle to make your subject appear more powerful.");
    }
    if (tip === "composition-astrophotography") {
        alert("Tip: Try to position the Milky Way in the frame to create a more interesting composition.");
    }
    if (tip === "lighting-astrophotography") {
        alert("Tip: Use a wide aperture to capture more light. Open your lens to f/2.8 or lower if possible.");
    }
    if (tip === "perspective-astrophotography") {
        alert("Tip: Use a telephoto lens to compress the perspective and achieve more detail in the stars. A 200mm lens or higher is recommended.");
    }
}

function showChallenge(type) {
    if (type === "landscape") {
        alert("Challenge: Take a photo of a landscape and share it with us!");
    }
    if (type === "astrophotography") {
        alert("Challenge: Take a photo of the night sky and share it with us!");
    }
}

function searchPhotos() {
    var input, filter, posts, i, txtValue;
    input = document.getElementById('search-input');
    filter = input.value.toUpperCase();
    posts = document.getElementsByClassName('post');

    for (i = 0; i < posts.length; i++) {
        txtValue = posts[i].textContent || posts[i].innerText;

        if (txtValue.toUpperCase().indexOf(filter) > -1) {
            posts[i].style.display = "flex";
        } else {
            posts[i].style.display = "none";
        }
    }
}

function toggleLike(button) {
    if (button.style.color == "red") {
        button.style.color = "black";
        button.innerHTML = "♡";
    }
    else {
        button.style.color = "red";
        button.innerHTML = "♥︎";
    }
}
